- 18:20 إِنَّهُمْ إِن يَظْهَرُوا۟ عَلَيْكُمْ يَرْجُمُوكُمْ أَوْ يُعِيدُوكُمْ فِى مِلَّتِهِمْ وَلَن تُفْلِحُوٓا۟ إِذًا أَبَدًۭا
- 18:21 وَكَذَٰلِكَ أَعْثَرْنَا عَلَيْهِمْ لِيَعْلَمُوٓا۟ أَنَّ وَعْدَ ٱللَّهِ حَقٌّۭ وَأَنَّ ٱلسَّاعَةَ لَا رَيْبَ فِيهَآ إِذْ يَتَنَٰزَعُونَ بَيْنَهُمْ أَمْرَهُمْ ۖ فَقَالُوا۟ ٱبْنُوا۟ عَلَيْهِم بُنْيَٰنًۭا ۖ رَّبُّهُمْ أَعْلَمُ بِهِمْ ۚ قَالَ ٱلَّذِينَ غَلَبُوا۟ عَلَىٰٓ أَمْرِهِمْ لَنَتَّخِذَنَّ عَلَيْهِم مَّسْجِدًۭا
- 18:22 سَيَقُولُونَ ثَلَٰثَةٌۭ رَّابِعُهُمْ كَلْبُهُمْ وَيَقُولُونَ خَمْسَةٌۭ سَادِسُهُمْ كَلْبُهُمْ رَجْمًۢا بِٱلْغَيْبِ ۖ وَيَقُولُونَ سَبْعَةٌۭ وَثَامِنُهُمْ كَلْبُهُمْ ۚ قُل رَّبِّىٓ أَعْلَمُ بِعِدَّتِهِم مَّا يَعْلَمُهُمْ إِلَّا قَلِيلٌۭ ۗ فَلَا تُمَارِ فِيهِمْ إِلَّا مِرَآءًۭ ظَٰهِرًۭا وَلَا تَسْتَفْتِ فِيهِم مِّنْهُمْ أَحَدًۭا
- 18:23 وَلَا تَقُولَنَّ لِشَا۟ىْءٍ إِنِّى فَاعِلٌۭ ذَٰلِكَ غَدًا
- 18:24 إِلَّآ أَن يَشَآءَ ٱللَّهُ ۚ وَٱذْكُر رَّبَّكَ إِذَا نَسِيتَ وَقُلْ عَسَىٰٓ أَن يَهْدِيَنِ رَبِّى لِأَقْرَبَ مِنْ هَٰذَا رَشَدًۭا
- 18:25 وَلَبِثُوا۟ فِى كَهْفِهِمْ ثَلَٰثَ مِا۟ئَةٍۢ سِنِينَ وَٱزْدَادُوا۟ تِسْعًۭا
- 18:26 قُلِ ٱللَّهُ أَعْلَمُ بِمَا لَبِثُوا۟ ۖ لَهُۥ غَيْبُ ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ ۖ أَبْصِرْ بِهِۦ وَأَسْمِعْ ۚ مَا لَهُم مِّن دُونِهِۦ مِن وَلِىٍّۢ وَلَا يُشْرِكُ فِى حُكْمِهِۦٓ أَحَدًۭا
- 18:27 وَٱتْلُ مَآ أُوحِىَ إِلَيْكَ مِن كِتَابِ رَبِّكَ ۖ لَا مُبَدِّلَ لِكَلِمَٰتِهِۦ وَلَن تَجِدَ مِن دُونِهِۦ مُلْتَحَدًۭا
- 18:28 وَٱصْبِرْ نَفْسَكَ مَعَ ٱلَّذِينَ يَدْعُونَ رَبَّهُم بِٱلْغَدَوٰةِ وَٱلْعَشِىِّ يُرِيدُونَ وَجْهَهُۥ ۖ وَلَا تَعْدُ عَيْنَاكَ عَنْهُمْ تُرِيدُ زِينَةَ ٱلْحَيَوٰةِ ٱلدُّنْيَا ۖ وَلَا تُطِعْ مَنْ أَغْفَلْنَا قَلْبَهُۥ عَن ذِكْرِنَا وَٱتَّبَعَ هَوَىٰهُ وَكَانَ أَمْرُهُۥ فُرُطًۭا
- 18:29 وَقُلِ ٱلْحَقُّ مِن رَّبِّكُمْ ۖ فَمَن شَآءَ فَلْيُؤْمِن وَمَن شَآءَ فَلْيَكْفُرْ ۚ إِنَّآ أَعْتَدْنَا لِلظَّٰلِمِينَ نَارًا أَحَاطَ بِهِمْ سُرَادِقُهَا ۚ وَإِن يَسْتَغِيثُوا۟ يُغَاثُوا۟ بِمَآءٍۢ كَٱلْمُهْلِ يَشْوِى ٱلْوُجُوهَ ۚ بِئْسَ ٱلشَّرَابُ وَسَآءَتْ مُرْتَفَقًا
- 18:30 إِنَّ ٱلَّذِينَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّٰلِحَٰتِ إِنَّا لَا نُضِيعُ أَجْرَ مَنْ أَحْسَنَ عَمَلًا
- 18:31 أُو۟لَٰٓئِكَ لَهُمْ جَنَّٰتُ عَدْنٍۢ تَجْرِى مِن تَحْتِهِمُ ٱلْأَنْهَٰرُ يُحَلَّوْنَ فِيهَا مِنْ أَسَاوِرَ مِن ذَهَبٍۢ وَيَلْبَسُونَ ثِيَابًا خُضْرًۭا مِّن سُندُسٍۢ وَإِسْتَبْرَقٍۢ مُّتَّكِـِٔينَ فِيهَا عَلَى ٱلْأَرَآئِكِ ۚ نِعْمَ ٱلثَّوَابُ وَحَسُنَتْ مُرْتَفَقًۭا
- 18:32 ۞ وَٱضْرِبْ لَهُم مَّثَلًۭا رَّجُلَيْنِ جَعَلْنَا لِأَحَدِهِمَا جَنَّتَيْنِ مِنْ أَعْنَٰبٍۢ وَحَفَفْنَٰهُمَا بِنَخْلٍۢ وَجَعَلْنَا بَيْنَهُمَا زَرْعًۭا
- 18:33 كِلْتَا ٱلْجَنَّتَيْنِ ءَاتَتْ أُكُلَهَا وَلَمْ تَظْلِم مِّنْهُ شَيْـًۭٔا ۚ وَفَجَّرْنَا خِلَٰلَهُمَا نَهَرًۭا
- 18:34 وَكَانَ لَهُۥ ثَمَرٌۭ فَقَالَ لِصَٰحِبِهِۦ وَهُوَ يُحَاوِرُهُۥٓ أَنَا۠ أَكْثَرُ مِنكَ مَالًۭا وَأَعَزُّ نَفَرًۭا
- 18:35 وَدَخَلَ جَنَّتَهُۥ وَهُوَ ظَالِمٌۭ لِّنَفْسِهِۦ قَالَ مَآ أَظُنُّ أَن تَبِيدَ هَٰذِهِۦٓ أَبَدًۭا
- 18:36 وَمَآ أَظُنُّ ٱلسَّاعَةَ قَآئِمَةًۭ وَلَئِن رُّدِدتُّ إِلَىٰ رَبِّى لَأَجِدَنَّ خَيْرًۭا مِّنْهَا مُنقَلَبًۭا
- 18:37 قَالَ لَهُۥ صَاحِبُهُۥ وَهُوَ يُحَاوِرُهُۥٓ أَكَفَرْتَ بِٱلَّذِى خَلَقَكَ مِن تُرَابٍۢ ثُمَّ مِن نُّطْفَةٍۢ ثُمَّ سَوَّىٰكَ رَجُلًۭا
- 18:38 لَّٰكِنَّا۠ هُوَ ٱللَّهُ رَبِّى وَلَآ أُشْرِكُ بِرَبِّىٓ أَحَدًۭا
- 18:39 وَلَوْلَآ إِذْ دَخَلْتَ جَنَّتَكَ قُلْتَ مَا شَآءَ ٱللَّهُ لَا قُوَّةَ إِلَّا بِٱللَّهِ ۚ إِن تَرَنِ أَنَا۠ أَقَلَّ مِنكَ مَالًۭا وَوَلَدًۭا
- 18:40 فَعَسَىٰ رَبِّىٓ أَن يُؤْتِيَنِ خَيْرًۭا مِّن جَنَّتِكَ وَيُرْسِلَ عَلَيْهَا حُسْبَانًۭا مِّنَ ٱلسَّمَآءِ فَتُصْبِحَ صَعِيدًۭا زَلَقًا
- 18:41 أَوْ يُصْبِحَ مَآؤُهَا غَوْرًۭا فَلَن تَسْتَطِيعَ لَهُۥ طَلَبًۭا
- 18:42 وَأُحِيطَ بِثَمَرِهِۦ فَأَصْبَحَ يُقَلِّبُ كَفَّيْهِ عَلَىٰ مَآ أَنفَقَ فِيهَا وَهِىَ خَاوِيَةٌ عَلَىٰ عُرُوشِهَا وَيَقُولُ يَٰلَيْتَنِى لَمْ أُشْرِكْ بِرَبِّىٓ أَحَدًۭا
- 18:43 وَلَمْ تَكُن لَّهُۥ فِئَةٌۭ يَنصُرُونَهُۥ مِن دُونِ ٱللَّهِ وَمَا كَانَ مُنتَصِرًا
- 18:44 هُنَالِكَ ٱلْوَلَٰيَةُ لِلَّهِ ٱلْحَقِّ ۚ هُوَ خَيْرٌۭ ثَوَابًۭا وَخَيْرٌ عُقْبًۭا
- 18:45 وَٱضْرِبْ لَهُم مَّثَلَ ٱلْحَيَوٰةِ ٱلدُّنْيَا كَمَآءٍ أَنزَلْنَٰهُ مِنَ ٱلسَّمَآءِ فَٱخْتَلَطَ بِهِۦ نَبَاتُ ٱلْأَرْضِ فَأَصْبَحَ هَشِيمًۭا تَذْرُوهُ ٱلرِّيَٰحُ ۗ وَكَانَ ٱللَّهُ عَلَىٰ كُلِّ شَىْءٍۢ مُّقْتَدِرًا
- 18:46 ٱلْمَالُ وَٱلْبَنُونَ زِينَةُ ٱلْحَيَوٰةِ ٱلدُّنْيَا ۖ وَٱلْبَٰقِيَٰتُ ٱلصَّٰلِحَٰتُ خَيْرٌ عِندَ رَبِّكَ ثَوَابًۭا وَخَيْرٌ أَمَلًۭا
- 18:47 وَيَوْمَ نُسَيِّرُ ٱلْجِبَالَ وَتَرَى ٱلْأَرْضَ بَارِزَةًۭ وَحَشَرْنَٰهُمْ فَلَمْ نُغَادِرْ مِنْهُمْ أَحَدًۭا
- 18:48 وَعُرِضُوا۟ عَلَىٰ رَبِّكَ صَفًّۭا لَّقَدْ جِئْتُمُونَا كَمَا خَلَقْنَٰكُمْ أَوَّلَ مَرَّةٍۭ ۚ بَلْ زَعَمْتُمْ أَلَّن نَّجْعَلَ لَكُم مَّوْعِدًۭا
- 18:49 وَوُضِعَ ٱلْكِتَٰبُ فَتَرَى ٱلْمُجْرِمِينَ مُشْفِقِينَ مِمَّا فِيهِ وَيَقُولُونَ يَٰوَيْلَتَنَا مَالِ هَٰذَا ٱلْكِتَٰبِ لَا يُغَادِرُ صَغِيرَةًۭ وَلَا كَبِيرَةً إِلَّآ أَحْصَىٰهَا ۚ وَوَجَدُوا۟ مَا عَمِلُوا۟ حَاضِرًۭا ۗ وَلَا يَظْلِمُ رَبُّكَ أَحَدًۭا
- 18:50 وَإِذْ قُلْنَا لِلْمَلَٰٓئِكَةِ ٱسْجُدُوا۟ لِءَادَمَ فَسَجَدُوٓا۟ إِلَّآ إِبْلِيسَ كَانَ مِنَ ٱلْجِنِّ فَفَسَقَ عَنْ أَمْرِ رَبِّهِۦٓ ۗ أَفَتَتَّخِذُونَهُۥ وَذُرِّيَّتَهُۥٓ أَوْلِيَآءَ مِن دُونِى وَهُمْ لَكُمْ عَدُوٌّۢ ۚ بِئْسَ لِلظَّٰلِمِينَ بَدَلًۭا
- 18:51 ۞ مَّآ أَشْهَدتُّهُمْ خَلْقَ ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ وَلَا خَلْقَ أَنفُسِهِمْ وَمَا كُنتُ مُتَّخِذَ ٱلْمُضِلِّينَ عَضُدًۭا
- 18:52 وَيَوْمَ يَقُولُ نَادُوا۟ شُرَكَآءِىَ ٱلَّذِينَ زَعَمْتُمْ فَدَعَوْهُمْ فَلَمْ يَسْتَجِيبُوا۟ لَهُمْ وَجَعَلْنَا بَيْنَهُم مَّوْبِقًۭا
- 18:53 وَرَءَا ٱلْمُجْرِمُونَ ٱلنَّارَ فَظَنُّوٓا۟ أَنَّهُم مُّوَاقِعُوهَا وَلَمْ يَجِدُوا۟ عَنْهَا مَصْرِفًۭا
- 18:54 وَلَقَدْ صَرَّفْنَا فِى هَٰذَا ٱلْقُرْءَانِ لِلنَّاسِ مِن كُلِّ مَثَلٍۢ ۚ وَكَانَ ٱلْإِنسَٰنُ أَكْثَرَ شَىْءٍۢ جَدَلًۭا
- 18:55 وَمَا مَنَعَ ٱلنَّاسَ أَن يُؤْمِنُوٓا۟ إِذْ جَآءَهُمُ ٱلْهُدَىٰ وَيَسْتَغْفِرُوا۟ رَبَّهُمْ إِلَّآ أَن تَأْتِيَهُمْ سُنَّةُ ٱلْأَوَّلِينَ أَوْ يَأْتِيَهُمُ ٱلْعَذَابُ قُبُلًۭا
- 18:56 وَمَا نُرْسِلُ ٱلْمُرْسَلِينَ إِلَّا مُبَشِّرِينَ وَمُنذِرِينَ ۚ وَيُجَٰدِلُ ٱلَّذِينَ كَفَرُوا۟ بِٱلْبَٰطِلِ لِيُدْحِضُوا۟ بِهِ ٱلْحَقَّ ۖ وَٱتَّخَذُوٓا۟ ءَايَٰتِى وَمَآ أُنذِرُوا۟ هُزُوًۭا
- 18:57 وَمَنْ أَظْلَمُ مِمَّن ذُكِّرَ بِـَٔايَٰتِ رَبِّهِۦ فَأَعْرَضَ عَنْهَا وَنَسِىَ مَا قَدَّمَتْ يَدَاهُ ۚ إِنَّا جَعَلْنَا عَلَىٰ قُلُوبِهِمْ أَكِنَّةً أَن يَفْقَهُوهُ وَفِىٓ ءَاذَانِهِمْ وَقْرًۭا ۖ وَإِن تَدْعُهُمْ إِلَى ٱلْهُدَىٰ فَلَن يَهْتَدُوٓا۟ إِذًا أَبَدًۭا
- 18:58 وَرَبُّكَ ٱلْغَفُورُ ذُو ٱلرَّحْمَةِ ۖ لَوْ يُؤَاخِذُهُم بِمَا كَسَبُوا۟ لَعَجَّلَ لَهُمُ ٱلْعَذَابَ ۚ بَل لَّهُم مَّوْعِدٌۭ لَّن يَجِدُوا۟ مِن دُونِهِۦ مَوْئِلًۭا
- 18:59 وَتِلْكَ ٱلْقُرَىٰٓ أَهْلَكْنَٰهُمْ لَمَّا ظَلَمُوا۟ وَجَعَلْنَا لِمَهْلِكِهِم مَّوْعِدًۭا
- 18:60 وَإِذْ قَالَ مُوسَىٰ لِفَتَىٰهُ لَآ أَبْرَحُ حَتَّىٰٓ أَبْلُغَ مَجْمَعَ ٱلْبَحْرَيْنِ أَوْ أَمْضِىَ حُقُبًۭا
- 18:61 فَلَمَّا بَلَغَا مَجْمَعَ بَيْنِهِمَا نَسِيَا حُوتَهُمَا فَٱتَّخَذَ سَبِيلَهُۥ فِى ٱلْبَحْرِ سَرَبًۭا
- 18:62 فَلَمَّا جَاوَزَا قَالَ لِفَتَىٰهُ ءَاتِنَا غَدَآءَنَا لَقَدْ لَقِينَا مِن سَفَرِنَا هَٰذَا نَصَبًۭا
- 18:63 قَالَ أَرَءَيْتَ إِذْ أَوَيْنَآ إِلَى ٱلصَّخْرَةِ فَإِنِّى نَسِيتُ ٱلْحُوتَ وَمَآ أَنسَىٰنِيهُ إِلَّا ٱلشَّيْطَٰنُ أَنْ أَذْكُرَهُۥ ۚ وَٱتَّخَذَ سَبِيلَهُۥ فِى ٱلْبَحْرِ عَجَبًۭا
- 18:64 قَالَ ذَٰلِكَ مَا كُنَّا نَبْغِ ۚ فَٱرْتَدَّا عَلَىٰٓ ءَاثَارِهِمَا قَصَصًۭا

# Güneşin battığı gözede bir halk

## Bir yolun ucunda

Ayet bir cümlenin ortasından açılır: {ar:حَتَّىٰٓ إِذَا بَلَغَ, tr:hattâ izâ belağa, gloss:sonunda ... vardığında}. Bu "sonunda" hemen önceki iki ayetten güç alır. Allah Zülkarneyn'e yeryüzünde imkân vermiş, ona her şeyden bir {ar:سَبَبًۭا, tr:sebeben, gloss:bir yere götüren yol ya da vasıta} vermiştir. O da bir sebebin ardına düşmüştür (18:84, 18:85). "Hattâ" bu izleyişin nerede durduğunu gösterir. Yol bir şehirde ya da bir düşmanın önünde bitmez. Bir ışığın gidişinde biter: {ar:مَغْرِبَ ٱلشَّمْسِ, tr:mağribe'ş-şems, gloss:güneşin battığı yer ya da vakit}.

Arapçada "mağrib" batışın hem yeri hem vaktidir. Aynı kök birkaç kelime sonra fiil olarak döner: {ar:وَجَدَهَا تَغْرُبُ, tr:vecedehâ tağrubu, gloss:onu batmakta buldu}. Varılan yer, orada olmakta olan şeyin adını taşır. Bu kökün temel resmi uzaklaşmak, gözden çekilip gitmektir. Sözlükte aynı aileden gelen "ğarîb" kelimesi uzakta kalmış, yabancı düşmüş kimse demektir. Zülkarneyn'in ulaştığı nokta, bir şeyin ayrılıp gittiği yerdir.

Ayetin ağırlığı bir fiilin iki kez kullanılmasında toplanır. Önce {ar:وَجَدَهَا, tr:vecedehâ, gloss:onu buldu}, yani güneşi batmakta buldu. Hemen ardından {ar:وَوَجَدَ عِندَهَا قَوْمًۭا, tr:ve vecede ındehâ kavmen, gloss:onun yanında bir halk buldu}. İlk "buldu" bir görüntüyü karşılar. Güneşin kara bir gözeye gömülüşü, yolun ucunda duran birinin gözüne göründüğü gibi anlatılır. Denizde yol alan birinin güneşi suya batıyor görmesi gibi, ayet de sahneyi Zülkarneyn'in durduğu yerden kurar. Aynı fiil ikinci kez gelince nesnesi değişir: bu sefer bulunan şey insanlardır. İlk bulunan, bulandan hiçbir şey istemez; güneş onun ne yaptığına bakmadan batar. İkinci bulunan ise bir karşılık bekler. Bulma bir seyirden sorumluluğa geçer. Bu surede "bulmak" zaten bir yüzleşme fiilidir. Kıyamet sahnesinde insanlar {ar:وَوَجَدُوا۟ مَا عَمِلُوا۟ حَاضِرًۭا, tr:ve vecedû mâ amilû hâdıran, gloss:yaptıklarını karşılarında hazır buldular} (18:49). Zülkarneyn'in yolun ucunda bulduğu halka ne yapacağı da ileride bulunacak şeylerin arasına girecektir.

## Kara balçıklı göze

Güneşin battığı yerin adı {ar:عَيْنٍ حَمِئَةٍۢ, tr:aynin hamietin, gloss:kara balçıklı bir göze}. "Ayn" suyun yerden çıktığı gözedir ve Arapçada aynı kelime "göz" demektir. Sahne bir bakışın sınırından anlatıldığı için bu ikinci anlam kulakta kalır: güneş bir "ayn"a gömülürken bir gözün ufku da orada kapanır. Surenin kendisi de bu kelimeyi göz anlamında kullanır. Peygambere {ar:وَلَا تَعْدُ عَيْنَاكَ عَنْهُمْ, tr:ve lâ ta'du aynâke anhum, gloss:gözlerin onları aşıp başka yere kaymasın} denir (18:28). Cehennemin sunulduğu gün de gözleri örtü içinde olanlardan söz edilir (18:101). Buradaki kelime gözeyi anlatır. Ama göz anlamı, bu sahnenin bir bakışla kurulduğunu hatırlatır.

"Hamie" sıfatı Kur'an'da yalnızca burada geçer. Aynı kökün ismi {ar:حْمًۭا, tr:hame', gloss:kararmış ve bozulmuş balçık} başka bir surede insanın yaratıldığı malzemenin adıdır (15:26, 15:28, 15:33). Orada insan, şekil verilmiş kara balçıktan kuruyup çınlayan bir çamurdan yaratılır. Burada ise gökteki en parlak cisim, aynı kökten bir sıfatla anılan bir balçığa iner. Hemen yanında da insanlar durur. İki pasaj arasında aynı kelime değil, aynı kök vardır. Bu yüzden bu bir yorum benzerliğidir, bir eşitleme değildir. Yine de dinleyici şunu duyabilir: yolun sonunda ışık, insanın yapıldığı türden bir karanlık malzemeye çekilir ve tam orada insanın ne yapacağı sorulur.

Kanonik kıraatlerin bir kısmı bu kelimeyi {ar:حَامِيَةٍ, tr:hâmiyetin, gloss:kızgın ya da sıcak} diye okur. O okuyuşta göze kaynar sulu bir kaynaktır. İki okuyuş birbirini dışlamaktan çok aynı yerin iki niteliğini verir. Biri balçığın karanlığını, öteki batan güneşin sıcaklığını öne çıkarır. Burada esas alınan metin karanlık balçığı söyler, ama sıcaklık okuyuşu bu gözenin durgun bir gölet olmadığını da hissettirir.

Güneşin bu surede insanlarla ilişkisi üç sahnede kurulur. Mağaradaki gençlerin başında güneş doğarken mağaralarından sağa kayar, {ar:وَإِذَا غَرَبَت تَّقْرِضُهُمْ ذَاتَ ٱلشِّمَالِ, tr:ve izâ ğarabet takrıduhum zâte'ş-şimâl, gloss:battığında da onları sol yandan kesip geçer} (18:17). Bu, bizim ayetteki batma fiilinin aynısıdır. Orada güneşin yolu insanların üzerinden çekilir; gençler bir kovukta, ışığın dokunmadığı bir aralıkta korunur. Doğuda ise Zülkarneyn güneşi {ar:تَطْلُعُ عَلَىٰ قَوْمٍۢ لَّمْ نَجْعَل لَّهُم مِّن دُونِهَا سِتْرًۭا, tr:tatluu alâ kavmin lem nec'al lehum min dûnihâ sitrâ, gloss:kendileriyle arasına hiçbir örtü koymadığımız bir halkın üzerine doğarken} bulur (18:90). Orada insanlar ışığa çıplak kalır. Batıda ise ışık gider, kara bir gözeye çekilir ve halk bu kayboluşun {ar:عِندَهَا, tr:ındehâ, gloss:yanında} durur. "Yanında" kelimesi halkı gözenin içine koymaz, kenarına yerleştirir. Kelimedeki dişil zamir hem güneşe hem gözeye dönebilir, ikisi de Arapçada dişildir. Böylece sahne katmanlı kalır: batan ışık ile kara su tek bir kıyıda birleşir, insanlar o kıyıdadır. Mağaradaki gençler ışıktan korunmuştu, doğudakiler ışığa açıktı. Bunlar ise ışığın çekildiği yerde yaşar ve hakkında karar verilecek kişilerdir.

## Musa'nın yolunun aynası

Bu kuruluş surede ilk kez burada görülmez. Musa da bir yola çıkarken {ar:لَآ أَبْرَحُ حَتَّىٰٓ أَبْلُغَ مَجْمَعَ ٱلْبَحْرَيْنِ, tr:lâ ebrahu hattâ ebluğa mecmea'l-bahreyn, gloss:iki denizin birleştiği yere varıncaya kadar durmayacağım} der (18:60). Aynı "hattâ", aynı "varmak" fiili, yine ikili bir yer adıyla kullanılır. Oraya vardıklarında balık {ar:فَٱتَّخَذَ سَبِيلَهُۥ فِى ٱلْبَحْرِ سَرَبًۭا, tr:fettehaze sebîlehû fi'l-bahri seraben, gloss:denizde bir tünel gibi yolunu tuttu} (18:61). Sonra {ar:فَوَجَدَا عَبْدًۭا مِّنْ عِبَادِنَآ, tr:fe vecedâ abden min ibâdinâ, gloss:kullarımızdan bir kul buldular} (18:65). Sıra aynıdır: varılır, bir şey suya geçip gider, ardından biri bulunur. Zülkarneyn'in sahnesinde de ulaşılır, güneş gözeye çekilir ve bir halk bulunur.

Ancak iki hikâyede roller ters çevrilmiştir. Musa yolun sonunda bir öğretmen bulur ve sabretmesi gereken taraf olur. Gemiyi delen, çocuğu öldüren, duvarı dikelten kişinin gerekçeleri ondan en sona kadar gizlenir (18:78, 18:82). Zülkarneyn ise yolun sonunda bir halk bulur ve eylemde bulunacak taraf olur. Üstelik gerekçesini eylemden önce açıkça söyler (18:87, 18:88). Musa çocuğun öldürülmesine {ar:لَّقَدْ جِئْتَ شَيْـًۭٔا نُّكْرًۭا, tr:lekad ci'te şey'en nukrâ, gloss:gerçekten çok çirkin ve tanınmaz bir iş yaptın} diyerek itiraz etmişti (18:74). Zülkarneyn'in cevabında aynı kelime geri gelir, ama o "nukr" azabı kendi eline almaz, Rabbe bırakır. Buna aşağıda döneceğiz.

## İki yol, iki ağırlık

Ardından ilahi hitap gelir: {ar:قُلْنَا يَٰذَا ٱلْقَرْنَيْنِ, tr:kulnâ yâ ze'l-karneyn, gloss:Dedik ki: Ey iki boynuz sahibi}. Kur'an bu unvanı açıklamaz. Eski bir yorum onu, güneşin iki "boynuzuna", yani doğuya ve batıya ulaşmış olmasıyla açıklar. Bu ayet ile 18:90 onun iki ucu gibidir. İkili bir adla çağrılan bu adam, iki yol arasında bırakılmak üzeredir. Bu sadece bir ses yankısıdır, ama seçimin kuruluşu da ikilidir: {ar:إِمَّآ أَن تُعَذِّبَ وَإِمَّآ أَن تَتَّخِذَ فِيهِمْ حُسْنًۭا, tr:immâ en tuazzibe ve immâ en tettehıze fîhim husnâ, gloss:ya azap edersin ya da onlar hakkında iyiliği tutum edinirsin}. İki "immâ" iki kefeyi aynı çerçeveye oturtur. Hiçbiri varsayılan yol değildir, ikisi de aynı dilbilgisi ağırlığıyla sunulur.

Ama kefeler aynı biçimde doldurulmamıştır. Birinci yol nesnesizdir: "azap edersin", kime olduğu söylenmez. Halk ancak ikinci yolda açıkça anılır: {ar:فِيهِمْ, tr:fîhim, gloss:onların içinde ya da onlar hakkında}. Metin, cezalandırma yolunda halkı adıyla hedef göstermez; halkı iyilik yolunda anar.

İkinci yolun fiili {ar:تَتَّخِذَ, tr:tettehıze, gloss:edinmek ya da bir şeyi bir şey olarak tutmak}. Bu fiil Kehf suresinde neredeyse hep yanlış edinmeler için kullanılır. Surenin başında uyarılanlar Allah'ın çocuk edindiğini söyleyenlerdir (18:4). Mağaradaki gençler halklarını şöyle anlatır: {ar:هَٰٓؤُلَآءِ قَوْمُنَا ٱتَّخَذُوا۟ مِن دُونِهِۦٓ ءَالِهَةًۭ, tr:hâulâi kavmunâ-ttehazû min dûnihî âliheten, gloss:şu kavmimiz O'ndan başka ilahlar edindi} (18:15). İblis ve soyu dost edinilir (18:50). Ayetler alay konusu edinilir (18:56, 18:106). Bir "kavim"in edindiği şeyle başlayan sure, burada bir kavmin karşısında duran güç sahibine neyi edinebileceğini söyler.

Bu seçimin arkasında surenin bir başka sorusu durur: gücü olan, bir halktan ne alır? Musa, misafir etmeyen köyün yıkılmak üzere olan duvarını dikelten arkadaşına {ar:لَوْ شِئْتَ لَتَّخَذْتَ عَلَيْهِ أَجْرًۭا, tr:lev şi'te letteḫazte aleyhi ecrâ, gloss:isteseydin bunun için bir ücret alırdın} der (18:77). Aynı fiil burada bir iş karşılığı almak demektir. Zülkarneyn'in üçüncü durağında bir halk ona {ar:فَهَلْ نَجْعَلُ لَكَ خَرْجًا, tr:fehel nec'alu leke harcen, gloss:sana bir vergi verelim mi} diye sorar (18:94). O da {ar:مَا مَكَّنِّى فِيهِ رَبِّى خَيْرٌۭ, tr:mâ mekkennî fîhi rabbî hayr, gloss:Rabbimin bana verdiği imkân daha hayırlıdır} diye cevap verir (18:95). Bizim ayette edinilecek şey bir ücret ya da vergi değildir. Halktan alınan bir şey de değildir. "Fîhim" iyiliği halkın içine, onlarla kurulacak ilişkinin kendisine yerleştirir.

Edinilecek şeyin adı {ar:حُسْنًۭا, tr:husnen, gloss:iyilik ve güzellik}. Bu kelime surenin başındaki sınavı geri çağırır: {ar:إِنَّا جَعَلْنَا مَا عَلَى ٱلْأَرْضِ زِينَةًۭ لَّهَا لِنَبْلُوَهُمْ أَيُّهُمْ أَحْسَنُ عَمَلًۭا, tr:innâ cealnâ mâ ale'l-ardı zîneten lehâ linebluvehum eyyuhum ahsenu amelâ, gloss:yeryüzündeki her şeyi ona süs yaptık ki hangisinin daha güzel iş yaptığını sınayalım} (18:7). Zülkarneyn'e {ar:إِنَّا مَكَّنَّا لَهُۥ فِى ٱلْأَرْضِ, tr:innâ mekkennâ lehû fi'l-ard, gloss:biz ona yeryüzünde imkân verdik} (18:84) denmişti. Yeryüzünde yerleşik kılınmış bu adamın sınavı, dünyanın ucunda iki yol arasında önüne konur. Seçim halk hakkında bir karar olduğu kadar, seçen hakkında da bir ölçüdür. Sure bu ölçünün tuzağını da gösterir: bazıları {ar:يَحْسَبُونَ أَنَّهُمْ يُحْسِنُونَ صُنْعًا, tr:yahsebûne ennehum yuhsinûne sun'â, gloss:güzel iş yaptıklarını sanırlar} (18:104). Bu fiil de aynı kökten gelir. İyilik, sanıldığı için değil yapıldığı için sayılır.

Seçimin ne kadar ağır olduğunu surenin Allah'ı anlatan bir ayeti gösterir: {ar:وَرَبُّكَ ٱلْغَفُورُ ذُو ٱلرَّحْمَةِ ۖ لَوْ يُؤَاخِذُهُم بِمَا كَسَبُوا۟ لَعَجَّلَ لَهُمُ ٱلْعَذَابَ, tr:ve rabbuke'l-ğafûru zu'r-rahmeh lev yuâhızuhum bimâ kesebû leaccele lehumu'l-azâb, gloss:Rabbin bağışlayandır ve rahmet sahibidir; kazandıkları yüzünden onları hemen sorgulasaydı azabı çabuklaştırırdı} (18:58). Allah azabı ertelemekte, bir vakte bırakmaktadır. Rahmetin sahibi için kullanılan "zû" (sahip) kelimesi, "iki boynuz sahibi" unvanındaki kelimenin aynısıdır. Bu bir kelime yankısıdır, iki "sahip" arasında denklik kurmaz. Ama sahneyi keskinleştirir. Rahmet sahibi Rab azabı erteler; iki boynuz sahibi bir insana ise azabı şimdi uygulama imkânı açılır.

## Cevap: nesnesiz azaba bir sınır

Zülkarneyn iki yoldan birini bütün halk için seçmez. Nesnesiz bırakılmış azaba bir nesne verir ve onu daraltır: {ar:أَمَّا مَن ظَلَمَ فَسَوْفَ نُعَذِّبُهُۥ ثُمَّ يُرَدُّ إِلَىٰ رَبِّهِۦ فَيُعَذِّبُهُۥ عَذَابًۭا نُّكْرًۭا, tr:emmâ men zaleme fesevfe nuazzibuhû summe yureddu ilâ rabbihî feyuazzibuhû azâben nukrâ, gloss:zulmedene gelince onu cezalandıracağız; sonra Rabbine döndürülür ve O da onu görülmemiş bir azapla cezalandırır} (18:87). Kavim bölünür, azap yalnızca zulmedeni bulur. Ölçü rastgele değildir. Birkaç ayet önce helak edilen memleketler için {ar:أَهْلَكْنَٰهُمْ لَمَّا ظَلَمُوا۟, tr:ehleknâhum lemmâ zalemû, gloss:zulmettiklerinde onları helak ettik} denmişti (18:59). Zülkarneyn ilahi yargının ölçütünü kendi yargısına ölçüt yapar. Kendi cezasını da geçici bir basamak olarak koyar: "sonra Rabbine döndürülür". Musa'nın bir insan eylemi için kullandığı "nukr" kelimesi burada yalnızca Rabbin azabının sıfatıdır. Görülmemiş, tanınmaz olan ceza insanın elinden alınıp Allah'a bırakılmıştır.

İkinci kefeyi de o doldurur: {ar:وَأَمَّا مَنْ ءَامَنَ وَعَمِلَ صَٰلِحًۭا فَلَهُۥ جَزَآءً ٱلْحُسْنَىٰ, tr:ve emmâ men âmene ve amile sâlihan felehû cezâeni'l-husnâ, gloss:iman edip iyi iş yapana gelince ona karşılık olarak en güzeli vardır} (18:88). Ayetteki belirsiz "husn" (bir iyilik) burada belirlenmiş ve derecelenmiş "el-husnâ"ya (en güzel olan) döner. Bu cümle surenin açılışındaki kitap tarifine benzer. Kitap, iyi işler yapan müminlere {ar:أَنَّ لَهُمْ أَجْرًا حَسَنًۭا, tr:enne lehum ecran hasenâ, gloss:onlar için güzel bir karşılık olduğunu} müjdelemek ve şiddetli bir azapla uyarmak için indirilmiştir (18:2). Peygamberler de {ar:مُبَشِّرِينَ وَمُنذِرِينَ, tr:mubeşşirîne ve munzirîn, gloss:müjdeleyiciler ve uyarıcılar} olarak gönderilir (18:56). Dünyanın batı ucunda bir yönetici, Kitabın iki işini, uyarmayı ve müjdelemeyi, bir halkın arasında uygular.

Cevabın son cümlesi Musa'nın yolundaki bir itirazı ters çevirir. Musa, sınırlarını aşan yol arkadaşından {ar:وَلَا تُرْهِقْنِى مِنْ أَمْرِى عُسْرًۭا, tr:ve lâ turhiknî min emrî usrâ, gloss:işimden dolayı bana zorluk yükleme} diye rica etmişti (18:73). Zülkarneyn iyilik yapana {ar:وَسَنَقُولُ لَهُۥ مِنْ أَمْرِنَا يُسْرًۭا, tr:ve senekûlu lehû min emrinâ yusrâ, gloss:ona buyruğumuzdan kolaylık söyleyeceğiz} der (18:88). Aynı kalıp, zorluk yerine kolaylık kelimesiyle gelir. Gücü elinde tutan kişi, iyilik yapan için yükü hafifletmeyi söz verir.

## Ayete yeniden bakınca

Artık ayet baştan okunduğunda iki "fî" (içinde) birbirine karşılık verir. Ayetin başında güneş {ar:فِى عَيْنٍ حَمِئَةٍۢ, tr:fî aynin hamietin, gloss:kara balçıklı bir gözenin içinde} batar. Ayetin sonunda iyilik {ar:فِيهِمْ, tr:fîhim, gloss:onların içinde} edinilir. Ayet, ışığın karanlık bir malzemeye çekildiği bir kıyıdan başlar ve iyiliğin bir halkın içine yerleştirilebileceği bir imkânla biter. Aradaki köprü ikinci "buldu"dur. Yolun sonunda asıl bulunan güneşin gidişi değil, onun yanında yaşayan insanlardır. Onlarla karşılaşılınca da ilahi söz, güç sahibine iki yol gösterir. Yollardan biri adı konmamış bir cezadır, öteki halkın adını anan bir iyiliktir. Zülkarneyn'in cevabı, ona verilen imkânın nasıl taşınacağını gösterir. Ceza zulmedenle sınırlanır ve son hüküm Rabbe bırakılır. İyilik ise iyilik yapana "en güzel" olarak geri döner.

# Güneşin battığı yerde bir seçenek

## Yolun ucu

Âyet, süren bir hareketin ortasında başlar. {ar:حَتَّىٰٓ إِذَا, tr:hattâ izâ, gloss:ta ki … olunca} kalıbı, önceki iki âyette açılan yolu bir bitiş noktasına bağlar. Zülkarneyn'e yeryüzünde imkân verilmiş, her şeyden ona bir yol ve vesile sağlanmıştır (18:84). O da bir yolu izlemiştir (18:85). Bu âyet o yolun nerede bittiğini söyler.

Fiil {ar:بَلَغَ, tr:belağa, gloss:vardı, ulaştı}'dır. Klasik sözlükler bu fiili sıradan bir "gitmek" diye tanımlamaz: {ar:البلوغ والبلاغ الانتهاء إلى أقصى المقصد والمنتهى مكانا كان أو زمانا أو أمرا, tr:el-bulûğ ve'l-belâğ el-intihâu ilâ aksa'l-maksadi ve'l-müntehâ mekânen kâne ev zemânen ev emrâ, gloss:ulaşmak, amaçlanan yerin ve sınırın en uç noktasına varmaktır; bu uç bir yer, bir zaman ya da bir iş olabilir}. Demek ki varılan yer bir durak değil, bir sınırdır.

Sınırın adı {ar:مَغْرِبَ ٱلشَّمْسِ, tr:mağribe'ş-şems, gloss:güneşin battığı yer}'dir. Bir sözlükçü kökün batış anlamını şöyle açıklar: {ar:غروب الشمس كأنه بعدها عن وجه الأرض, tr:ğurûbu'ş-şems ke-ennehû bu'duhâ an vechi'l-ard, gloss:güneşin batışı, sanki onun yeryüzünden uzaklaşmasıdır}. Bu açıklamayla sahnenin tuhaflığı ortaya çıkar. Zülkarneyn uzaklaşan bir şeyin peşine düşmüş ve uzaklaşmanın gerçekleştiği yere kadar gitmiştir. Âyet aynı kökü hemen ikinci kez kullanır. Önce yeri bir adla, {ar:مَغْرِبَ, tr:mağrib, gloss:batış yeri} diye anar. Orada gördüğünü ise bir fiille anlatır: {ar:تَغْرُبُ, tr:teğrubu, gloss:batıyor}. Böylece haritadaki bir ad, gözün önünde olup biten bir olaya dönüşür.

Bu dönüşümü taşıyan fiil {ar:وَجَدَهَا, tr:vecedehâ, gloss:onu buldu}'dur. Sözlükler bulmanın bir türünü açıkça duyulara bağlar: {ar:وجود بإحدى الحواس الخمس, tr:vucûdun bi-ihde'l-havâssi'l-hams, gloss:beş duyudan biriyle bulmak}. Cümle, güneşin evrendeki yeri hakkında bir hüküm vermez. Yolun ucunda duran adamın gözünün neyle karşılaştığını bildirir; kıyıda duran birinin güneşi denize gömülürken görmesi gibi. Anlatılan şey bir gözlemcinin sınırda gördüğüdür ve âyet birazdan o gözlemciye bir soru yöneltecektir.

Sûre bu kalıbı daha önce de kullanmıştı. Mûsâ yola çıkarken şöyle der: {ar:لَآ أَبْرَحُ حَتَّىٰٓ أَبْلُغَ مَجْمَعَ ٱلْبَحْرَيْنِ, tr:lâ ebrahu hattâ ebluğa mecma'a'l-bahreyn, gloss:iki denizin birleştiği yere varıncaya kadar durmayacağım} (18:60). Sonra {ar:فَلَمَّا بَلَغَا مَجْمَعَ بَيْنِهِمَا, tr:fe-lemmâ beleğâ mecma'a beynihimâ, gloss:ikisi o birleşme yerine varınca} gelir. Balık denizde gözden kaybolur (18:61) ve ardından {ar:فَوَجَدَا عَبْدًۭا مِّنْ عِبَادِنَآ, tr:fe-vecedâ abden min ibâdinâ, gloss:kullarımızdan bir kul buldular} (18:65). Aynı fiiller aynı sırayla döner: önce bir yere varılır, sonra bir şey suya geçip kaybolur, sonra bir insan bulunur. Zülkarneyn'de de önce varış gelir, ardından güneş suya iner, ardından bir topluluk bulunur. Kelimeler aynıdır; sahnelerin birbirine benzemesi ise bir yorum benzerliğidir.

Bu benzerlik bir farkı keskinleştirir. Mûsâ sınırda kendisinden fazlasını bilen birini bulur ve öğrenci olur. Zülkarneyn sınırda hakkında hiçbir şey bilmediği bir topluluk bulur ve karar veren olur. Mûsâ'nın hikâyesi, gördüğü işleri hemen yargılayıp sabredemeyen bir adamın hikâyesiydi. Zülkarneyn'in hikâyesinde ise yargı yetkisi doğrudan adamın eline verilir.

## Kara balçıklı göz

Güneşin battığı yer {ar:عَيْنٍ حَمِئَةٍۢ, tr:aynin hamietin, gloss:kara balçıklı bir göze, pınar}'dır. Ayn Arapçada hem gözü hem yerden kaynayan suyu adlandırır. Sözlükler pınar anlamını gözle kurulan benzerliğe bağlar ve kaynaktan akan suyu {ar:ماء معين أي ظاهر للعيون, tr:mâun maînun ey zâhirun li'l-uyûn, gloss:maîn su, gözlere görünen sudur} diye açıklar. Aynı kelime güneşin kendisi için de kullanılır: {ar:عين الشمس مشبه بعين الإنسان, tr:aynu'ş-şems müşebbehun bi-ayni'l-insân, gloss:güneşin "göz"ü, insan gözüne benzetilerek böyle anılır}. Araplar {ar:طلعت العين وغابت العين، أي الشمس, tr:tala'ati'l-aynu ve gâbeti'l-aynu, eyi'ş-şems, gloss:"göz doğdu, göz battı", yani güneş} der.

Âyet burada güneşi şems diye anar, ayn diye değil. Bu yüzden ikinci resim âyetin sözü değil, dilin içinde hazır bekleyen bir resimdir. Yine de sahnenin yapısını açıklar. Bulma fiili bir bakışı anlatır. Bakılan şey, gökte "göz" diye de anılabilen bir ışıktır. O ışığın indiği yer ise yeryüzünün bir "göz"üdür. Sahne baştan sona görme üzerine kuruludur.

Ne var ki bu göz berrak değildir. Sözlükler {ar:حَمِئَةٍۢ, tr:hamietin} sıfatının dayandığı maddeyi açıkça tanımlar: {ar:الحمأ الطين الأسود المنتن, tr:el-hame'u't-tînu'l-esvedu'l-münten, gloss:hame', kara ve kokuşmuş balçıktır}. Bu âyetteki ifadeyi de ayrıca açıklarlar: {ar:عين حمئة أي ذات حمأة, tr:aynun hamietun ey zâtu hame'e, gloss:hamie göze, içinde kara balçık bulunan pınardır}. Kuyuculukta aynı kökten fiiller kurulur: bir kalıp kuyunun dibindeki balçığı çıkarmayı, başka bir kalıp kuyuya balçık atıp onu bozmayı anlatır. Kelime, suyun kaynağını tıkayan ve dibe çöken bir tortuyu adlandırır; rengiyle ve kokusuyla duyuları iten bir tortuyu. Kabul gören kıraatlerin bir kısmı kelimeyi "sıcak" anlamında hâmiye diye okur. O okuyuşta da pınar serin, durgun bir ayna değildir. İki okuyuşta da ışık parlak bir ufka değil, koyu ve yoğun bir suya iner.

Bu tortunun Kur'an'da başka bir yükü daha vardır. Aynı kökten gelen hame' adı, insanın yaratıldığı çamurun kara balçıktan geldiğini anlatan âyetlerde geçer (15:26, 15:28, 15:33). Kehf de insanın başlangıcını toprağa bağlar. İki bahçe sahibine arkadaşı şöyle der: {ar:أَكَفَرْتَ بِٱلَّذِى خَلَقَكَ مِن تُرَابٍۢ ثُمَّ مِن نُّطْفَةٍۢ ثُمَّ سَوَّىٰكَ رَجُلًۭا, tr:e-kefarte bi'llezî halakake min turâbin summe min nutfetin summe sevvâke racülâ, gloss:seni topraktan, sonra bir damla sudan yaratan, sonra seni düzgün bir insan kılanı mı inkâr ediyorsun?} (18:37). Buradaki bağ bir kök akrabalığına ve sûre içi bir temaya dayanır; âyet insanlardan çamur diye söz etmez. Ama şu duyulabilir: yeryüzünün batı ucunda, ışığın gömüldüğü yerde insanın ham maddesinin adı geçer ve hemen yanında bir insan topluluğu durur. Sahne gösterişli bir dünya sonu manzarası olmaktan çok, insanın kendi başlangıcına yakın bir kıyıdır.

{ar:وَوَجَدَ عِندَهَا قَوْمًۭا, tr:ve vecede indehâ kavmâ, gloss:ve onun yanında bir topluluk buldu}. Bulma fiili yeniden başlar ve bu sefer gözün nesnesi değişir: gökteki cisimden insanlara geçilir. Sözlükler {ar:عِندَهَا, tr:indehâ} içindeki kelimeyi {ar:لفظ موضوع للقرب, tr:lafzun mevdûun li'l-kurb, gloss:yakınlık bildirmek için konmuş bir kelime} diye tanımlar. Topluluk pınarın içinde değil, yanındadır. Zamir dişildir ve hem güneşe hem pınara gidebilir, çünkü Arapçada ikisi de dişildir. Hangisi seçilirse seçilsin insanlar aynı kıyıda durur: ışığın battığı yerin hemen berisinde yaşayan bir topluluk. {ar:قَوْمًۭا, tr:kavmen} belirsiz bırakılmıştır. Adları, dinleri, işleri söylenmez. Âyet onların iyi mi kötü mü olduğunu bildirmeden sözü doğrudan karara geçirir.

Zülkarneyn'in üç varışının üçünde de bulunan bir topluluktur ve her biri bir sınıra göre yerleştirilir. Batıda topluluk batışın yanındadır. Doğuda {ar:وَجَدَهَا تَطْلُعُ عَلَىٰ قَوْمٍۢ لَّمْ نَجْعَل لَّهُم مِّن دُونِهَا سِتْرًۭا, tr:vecedehâ tatlu'u alâ kavmin lem nec'al lehüm min dûnihâ sitrâ, gloss:onu, kendileriyle güneş arasına bir örtü koymadığımız bir topluluğun üzerine doğarken buldu} (18:90). Üçüncü varışta ise {ar:وَجَدَ مِن دُونِهِمَا قَوْمًۭا لَّا يَكَادُونَ يَفْقَهُونَ قَوْلًۭا, tr:vecede min dûnihimâ kavmen lâ yekâdûne yefkahûne kavlâ, gloss:iki setin berisinde neredeyse hiçbir söz anlamayan bir topluluk buldu} (18:93). Birinci topluluk yakınlığıyla, ikincisi korunmasızlığıyla, üçüncüsü sözden kopukluğuyla tanımlanır. Her seferinde de aynı soru açılır: gücü olan bir adam bu insanlara ne yapacak?

Güneşin batışı bu sûrede ilk kez burada anılmaz. Mağara gençleri anlatılırken aynı fiil kullanılmıştı: {ar:وَإِذَا غَرَبَت تَّقْرِضُهُمْ ذَاتَ ٱلشِّمَالِ, tr:ve izâ ğarabet takriduhum zâte'ş-şimâl, gloss:battığında da onları sol yandan geçip bırakırdı} (18:17). Orada batan güneş insanlara dokunmadan geçer. Gençler mağaranın açıklığında korunur ve âyet bunu {ar:ذَٰلِكَ مِنْ ءَايَٰتِ ٱللَّهِ, tr:zâlike min âyâtillâh, gloss:bu, Allah'ın âyetlerindendir} diye adlandırır. Güneşle insan arasındaki mesafeyi Allah doğrudan düzenler. Batının ucunda ise böyle bir düzenlemeden söz edilmez. Güneş iner, insanlar yanında durur ve söz oraya kadar gelmiş bir adama döner. Batışın kıyısındaki insanlara ne olacağı, mağarada Allah'ın elindeydi; burada bir insanın seçimine bırakılır.

## "Ey Zülkarneyn"

{ar:قُلْنَا, tr:kulnâ, gloss:dedik} ile anlatıcı "Biz" konuşan kişi olur. Bu, 18:84'te {ar:إِنَّا مَكَّنَّا لَهُۥ فِى ٱلْأَرْضِ, tr:innâ mekkennâ lehû fi'l-ard, gloss:Biz ona yeryüzünde imkân ve yer verdik} diyen "Biz"in ta kendisidir. İmkânı veren ile imkânın ucunda soruyu soran aynıdır. Güç, sınıra vardığında bir hitapla karşılaşır.

Hitap {ar:يَٰذَا ٱلْقَرْنَيْنِ, tr:yâ ze'l-karneyn, gloss:ey iki karn sahibi} biçimindedir. Karn'ın en bilinen anlamı boynuzdur, yani baştan çıkan sert ve güçlü çıkıntı. Sözlükler bu adı boynuza benzeyen başka şeylere de verir ve bunların arasında güneşin ufukta ilk görünen üst kenarını da sayar. Lakabı iki çıkıntıyla ya da iki uçla açıklayan görüşleri de aktarırlar. Âyet lakabı açıklamaz. Ama iki uçlu bir adla çağrılan adam güneşin yolunun bir ucunda durmaktadır ve öbür ucuna da gidecektir (18:90). Şimdi önüne iki kollu bir söz konur.

{ar:إِمَّآ أَن تُعَذِّبَ وَإِمَّآ أَن تَتَّخِذَ فِيهِمْ حُسْنًۭا, tr:immâ en tuazzibe ve immâ en tettehıze fîhim husnâ, gloss:ya azap edersin ya da onların içinde güzel olanı edinirsin}. Dışarıdan bakınca iki kol birbirinin aynısıdır. İkisi de immâ ile açılır, ikisinde de en edatı bir eylemi seçilebilecek bir seçenek hâline getirir. İçeriden bakınca simetri bozulur. İlk kolda fiil tek başınadır. {ar:تُعَذِّبَ, tr:tuazzibe, gloss:azap edersin, cezalandırırsın} kime yöneldiğini söylemez, çünkü nesnesi bellidir: o topluluk. İkinci kolda topluluk nesne olmaktan çıkar ve bir alana dönüşür: {ar:فِيهِمْ, tr:fîhim, gloss:onların içinde, onlar hakkında}. Nesnenin yerini artık insanlar değil {ar:حُسْنًۭا, tr:husnen, gloss:güzellik, iyilik} alır. Ceza doğrudan insanların üzerine iner, iyilik ise onların arasına kurulur. Âyette fî edatı iki kez geçer. Güneş bir pınarın içine batar; iyilik bir topluluğun içine yerleştirilir. Edat aynıdır, içine aldığı şey değişir.

## Güzel olanı edinmek

İkinci kolun fiili dikkat ister. {ar:تَتَّخِذَ, tr:tettehıze} "yapmak" ya da "davranmak" demek değildir. "Almak" kökünün, eylemi yapanın kendisine dönen bir kalıbıdır. Sözlükler bunu açıkça yazar: {ar:الاتخاذ افتعال من الأخذ, tr:el-ittihâzu ifti'âlün mine'l-ahz, gloss:ittihâz, "almak" fiilinden türemiş, eylemi yapana dönen bir kalıptır}. Kökün aslını da {ar:الأصل حوز الشيء وجبيه وجمعه, tr:el-aslu havzu'ş-şey'i ve cebyuhû ve cem'uhû, gloss:kökün aslı, bir şeyi kendi alanına katıp toplamaktır} diye verirler. İttihâz, bir şeyi kendi alanına çekip kendine ait bir düzen ya da tutum hâline getirmektir. Öyleyse teklif şudur: Bu insanların arasında kendine edineceğin, sahipleneceğin şey iyilik olsun.

Kehf bu fiili baştan sona bir sınav fiili olarak kullanır. Mağara gençleri kendi topluluklarını şöyle anlatır: {ar:هَٰٓؤُلَآءِ قَوْمُنَا ٱتَّخَذُوا۟ مِن دُونِهِۦٓ ءَالِهَةًۭ, tr:hâulâi kavmunâ'ttehazû min dûnihî âliheh, gloss:şu kavmimiz O'nu bırakıp ilahlar edindi} (18:15). Sûre aynı fiille başkalarını da anar: Allah'ın çocuk edindiğini söyleyenleri (18:4), İblis'i ve soyunu dost edinenleri (18:50), âyetleri alay konusu edinenleri (18:56, 18:106), Allah'ın kullarını O'ndan başka veli edinebileceklerini sananları (18:102). Neredeyse her seferinde fiilin nesnesi yanlış bir şeydir: tanrılar, dostlar, alay. Allah kendisi hakkında bile {ar:وَمَا كُنتُ مُتَّخِذَ ٱلْمُضِلِّينَ عَضُدًۭا, tr:ve mâ kuntu muttehize'l-mudıllîne adudâ, gloss:saptıranları yardımcı edinecek değildim} der (18:51). Bu dizinin içinde 18:86 fiile, sûrenin anlattığı yanlış edinmelerin karşısında duran bir nesne verir: husn.

Aynı kökün en yalın hâli, hemen önceki kıssada güç sahibi bir figürle görünür. Hızır gemiyi neden deldiğini anlatırken şöyle der: {ar:وَكَانَ وَرَآءَهُم مَّلِكٌۭ يَأْخُذُ كُلَّ سَفِينَةٍ غَصْبًۭا, tr:ve kâne verâehum melikün ye'huzu külle sefînetin ğasbâ, gloss:onların ilerisinde her gemiyi zorla alan bir hükümdar vardı} (18:79). Burada kök yalın "almak" kalıbındadır ve gücün almasının en kaba biçimini gösterir: gördüğü her gemiye el koyan bir hükümdar. Zülkarneyn de bir hükümdardır; üstelik ona "her şeyden" vesile verilmiştir. Kimsenin tanımadığı bir topluluğun yanında, dünyanın kenarında durmaktadır. Önüne aynı kökten bir fiil konur, ama bu kez alınacak şey insanların malı değil, iyiliktir.

Ücret meselesi de sûrede zaten açılmıştır. Mûsâ, kendilerini misafir etmeyen köyde duvarı doğrultan arkadaşına {ar:لَوْ شِئْتَ لَتَّخَذْتَ عَلَيْهِ أَجْرًۭا, tr:lev şi'te le'ttehazte aleyhi ecrâ, gloss:isteseydin buna karşılık bir ücret alırdın} der (18:77). Zülkarneyn üçüncü durakta tam da bu durumla karşılaşır. Topluluk ona haraç önerir (18:94). O ise {ar:مَا مَكَّنِّى فِيهِ رَبِّى خَيْرٌۭ, tr:mâ mekkennî fîhi rabbî hayr, gloss:Rabbimin bana verdiği imkân daha hayırlıdır} diye karşılık verir (18:95). Gücünü, 18:84'te Allah'ın kullandığı fiille anar. Batıda önüne konan "iyiliği edinme" seçeneği, iki set arasında bir tutuma dönüşmüştür: gücün kendine aldığı şey para değildir.

Kökün bir kalıbı sûrede cezanın yanında da durur: {ar:وَرَبُّكَ ٱلْغَفُورُ ذُو ٱلرَّحْمَةِ ۖ لَوْ يُؤَاخِذُهُم بِمَا كَسَبُوا۟ لَعَجَّلَ لَهُمُ ٱلْعَذَابَ, tr:ve rabbüke'l-ğafûru zü'r-rahme, lev yuâhızuhum bimâ kesebû le-accele lehumu'l-azâb, gloss:Rabbin bağışlayandır, rahmet sahibidir; onları kazandıkları yüzünden hesaba çekseydi azabı hemen verirdi} (18:58). Burada "almak" kökü, bu kez hesaba çekmek anlamını taşıyan bir kalıpta, azapla aynı cümlededir. Allah bunu yapmaz, azabı erteler. 18:86'da ise kök öbür tarafa geçmiştir. Azap bir kolda tek başına durur; "almak" kökü öbür kolda iyiliğin fiili olur. Rabbi niteleyen {ar:ذُو ٱلرَّحْمَةِ, tr:zü'r-rahme, gloss:rahmet sahibi} ifadesi de Zülkarneyn'in lakabındaki "sahibi" kelimesini taşır. Rahmet Sahibi'nin kendi davranışında görülen tavır, yani cezayı erteleyip bağışlamayı öne almak, iki karn sahibine seçeneklerden biri olarak sunulmuştur.

Peki edinilecek husn nedir? Sözlüklerin tanımı yalındır: {ar:الحسن ضد القبح, tr:el-husnu diddu'l-kubh, gloss:husn, çirkinliğin karşıtıdır}. Bir başka sözlük onu {ar:كل مبهج مرغوب فيه, tr:küllü mubhicin merğûbün fîh, gloss:sevindiren ve istenen her şey} diye tanımlar ve bu beğeninin akla, gönlün eğilimine ya da duyulara dayanabileceğini ekler. Husn yalnız ahlaki bir iyilik değildir; göze ve gönle hoş gelen güzelliği de kapsar. Bu tanım âyetin başındaki balçığın tanımının karşısında durur. Hame' iki duyuyla tarif edilmişti: gözü rahatsız eden karalık ve burnu iten koku. Âyet koyu ve kokulu bir su kaynağıyla başlar, duyuya ve akla hoş gelen şeyin adıyla biter. İkisinin arasında bir insanın vereceği karar vardır. Husnen belirsiz bırakılmıştır. Hangi iyiliğin kastedildiği söylenmez; yalnız neyin cinsinden olduğu söylenir.

Kelime, sûrenin en başındaki sınava bağlanır: {ar:إِنَّا جَعَلْنَا مَا عَلَى ٱلْأَرْضِ زِينَةًۭ لَّهَا لِنَبْلُوَهُمْ أَيُّهُمْ أَحْسَنُ عَمَلًۭا, tr:innâ cealnâ mâ ale'l-ardı zîneten lehâ li-nebluvehum eyyuhum ahsenu amelâ, gloss:yeryüzündeki şeyleri ona süs kıldık ki hangisinin daha güzel iş yaptığını sınayalım} (18:7). Ahsen, husn ile aynı köktendir. Yeryüzündeki her şey süs olarak konmuştur ve bu süs, insanlardan hangisinin daha güzel iş yapacağını sınamak içindir. Hemen ardından o yüzeyin bir gün çıplak toprağa döneceği söylenir (18:8). Zülkarneyn'e yeryüzünde imkân verilmiştir (18:84) ve o, yeryüzünün batı ucuna, "yeryüzündeki şeylerin" bittiği yere kadar gelmiştir. Orada sorulan soru, sûrenin başındaki sorunun ta kendisidir; yalnız artık belli bir adla ve belli bir toplulukla somutlaşmıştır.

Sûre bu güzelliği yanlışlıkla sahiplenmenin kolay olduğunu da bilir. Amelleri boşa gidenler hakkında şöyle der: {ar:وَهُمْ يَحْسَبُونَ أَنَّهُمْ يُحْسِنُونَ صُنْعًا, tr:ve hum yahsebûne ennehum yuhsinûne sun'â, gloss:üstelik güzel iş yaptıklarını sanırlar} (18:104). Husn edinmek, iyilik yaptığını sanmak demek değildir. Sınanan bir iştir.

## Seçenekten ayrıma

Zülkarneyn'in cevabı teklifin biçimini değiştirir (18:87, 18:88). Allah ona immâ … ve immâ demişti: "ya şu, ya bu." O ise emmâ … ve emmâ diye cevap verir: {ar:أَمَّا مَن ظَلَمَ, tr:emmâ men zaleme, gloss:zulmedene gelince} … {ar:وَأَمَّا مَنْ ءَامَنَ وَعَمِلَ صَٰلِحًۭا, tr:ve emmâ men âmene ve amile sâlihâ, gloss:iman edip iyi iş yapana gelince}. İki edat yalnızca ilk sesin harekesiyle ayrılır, ama işleri farklıdır. İmmâ, seçecek olanın önüne iki yol koyar. Emmâ ise karşıdaki insanları sınıflara ayırır. Zülkarneyn bütün topluluk için tek bir yol seçmez. Belirsiz bırakılmış topluluğu, insanların kendi yaptıklarına göre böler. Azap zulmedenin, güzellik iman edip iyi iş yapanın payı olur.

Bu dil sûrede yeni değildir. Hızır da gizli sebepleri aynı edatla açıklamıştı: {ar:أَمَّا ٱلسَّفِينَةُ, tr:emme's-sefînetu, gloss:gemiye gelince} (18:79), {ar:وَأَمَّا ٱلْغُلَٰمُ, tr:ve emme'l-ğulâmu, gloss:çocuğa gelince} (18:80), {ar:وَأَمَّا ٱلْجِدَارُ, tr:ve emme'l-cidâru, gloss:duvara gelince} (18:82). Orada emmâ, dışarıdan kötülük gibi görünen işlerin ardındaki ayrımı sonradan açıklıyordu. Burada ise bir kararın ölçüsünü karar verilmeden önce ilan eder. Mûsâ'nın kıssasında ayrım sonradan öğrenilir; Zülkarneyn'inkinde karar anında konur.

Cevabın ceza kolu kendi sınırını da bilir: {ar:فَسَوْفَ نُعَذِّبُهُۥ ثُمَّ يُرَدُّ إِلَىٰ رَبِّهِۦ فَيُعَذِّبُهُۥ عَذَابًۭا نُّكْرًۭا, tr:fe-sevfe nuazzibuhû summe yuraddu ilâ rabbihî fe-yuazzibuhû azâben nukrâ, gloss:onu cezalandıracağız, sonra Rabbine döndürülecek ve O da onu görülmemiş bir azapla cezalandıracak}. Zülkarneyn'in cezası son söz değildir. Zalim Rabbine döndürülecek ve asıl azap orada verilecektir. Nukr kelimesi sûrede daha önce Mûsâ'nın ağzından çıkmıştı. Çocuğun öldürüldüğünü görünce {ar:لَّقَدْ جِئْتَ شَيْـًۭٔا نُّكْرًۭا, tr:lekad ci'te şey'en nukrâ, gloss:gerçekten çok kötü, görülmemiş bir iş yaptın} demişti (18:74). Mûsâ korkunç görünen bir işi hemen yargılamıştı. Zülkarneyn ise korkunç azabı kendi eline almaz, Rabbin eline bırakır. Bu, sûrenin {ar:وَلَا يُشْرِكُ فِى حُكْمِهِۦٓ أَحَدًۭا, tr:ve lâ yuşriku fî hukmihî ehadâ, gloss:hükmüne kimseyi ortak etmez} (18:26) cümlesinin bir hükümdarın dilindeki karşılığıdır. Kendisine seçenek verilmiş olan adam bile hükmün sonunu kendine ait saymaz.

İyilik kolu ise tekliftaki kelimeyi geri döndürür. Allah "onların içinde husn edinmeyi" önermişti. Zülkarneyn iman edip iyi iş yapana {ar:فَلَهُۥ جَزَآءً ٱلْحُسْنَىٰ, tr:fe-lehû cezâen'il-husnâ, gloss:ona karşılık olarak en güzeli vardır} der. Belirsiz husn, belirli ve en üst derecedeki el-husnâ'ya dönüşür. Önerilen güzellik, verilen karşılığın adı olur. Ardından {ar:وَسَنَقُولُ لَهُۥ مِنْ أَمْرِنَا يُسْرًۭا, tr:ve se-nekûlu lehû min emrinâ yusrâ, gloss:ona işimizden, buyruğumuzdan kolay olanı söyleyeceğiz} der. Bu cümle sûrenin iki eski sesine karşılık verir. Mağara gençleri {ar:وَهَيِّئْ لَنَا مِنْ أَمْرِنَا رَشَدًۭا, tr:ve heyyi' lenâ min emrinâ raşedâ, gloss:işimizden bize bir doğruluk hazırla} diye dua etmişti (18:10). Mûsâ da Hızır'a {ar:وَلَا تُرْهِقْنِى مِنْ أَمْرِى عُسْرًۭا, tr:ve lâ turhiknî min emrî usrâ, gloss:bu işimden dolayı bana zorluk yükleme} demişti (18:73). Yusr ile usr ayrı köklerden gelir; ortada aynı kelimenin tekrarı değil, bir karşıtlık vardır. Mûsâ'nın kaçınmak istediği zorluğun karşısına, Zülkarneyn'in iyiye vaat ettiği kolaylık konur. Gençlerin Allah'tan istediği "işimizden bir hazırlık", bir hükümdarın ağzında iyilere verilen bir söze dönüşür.

Şimdi âyete baştan dönülebilir. Her şeyden vesile verilmiş bir adam, güneşin "yeryüzünden uzaklaştığı" yere kadar gider. Orada gördüğü bir zafer tablosu değildir. Işık, insanın ham maddesinin adını taşıyan koyu ve kokulu bir pınara iner. Hemen yanında adı bilinmeyen bir topluluk vardır. Mağarada güneşi insanlardan uzak tutan Allah burada doğrudan konuşur ve kararı adama bırakır. Kararın iki kolu dışarıdan dengeli görünür, ama cümle onları aynı biçimde kurmaz. Ceza insanların üzerine iner; güzellik insanların içine yerleştirilir. Bu ikinci kol, sûrenin baştan beri yanlış nesnelere bağladığı "edinme" fiiline doğru nesneyi verir. Zülkarneyn'in cevabı da bu ağırlığı taşır. İnsanları tek parça bir kitle gibi cezalandırmaz, yaptıklarına göre ayırır ve cezanın son kısmını Rabbine bırakır. Yolun ucu bir manzaranın değil, bir sınavın yeridir. Yeryüzünün süsünün bittiği kenarda, sûrenin başından beri sorulan "hangisi daha güzel iş yapacak?" sorusu en çok imkâna sahip adamın önüne konur.
